```python
# 修复方向 B：契约检查改为 ast 节点定位，不依赖行号切片
import ast
import inspect


def _func_src(obj, name: str) -> str:
    mod_src = inspect.getsource(inspect.getmodule(obj))
    tree = ast.parse(mod_src)
    for node in ast.walk(tree):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)) and node.name == name:
            return ast.get_source_segment(mod_src, node) or ""
    return ""


src = _func_src(Orchestrator, "_should_reject")
assert "semantic_result is not None" in src
```
